# Image Scale Bar Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Calibrate the image by drawing a line of known length, then draw a map-style scale bar that snaps to a round value and re-labels itself as the view zooms.

**Architecture:** A Qt-free `scale_bar.py` owns the units and the snapping ladder. The calibration is a `ScaleCalibration` on the analysis session, persisted in the experiment's existing `roi_config.json` and seeded from preferences on first use. A small `ScaleCanvasLayer` handles the calibration drag, offered the mouse before the ROI layer, and `_ImageView.drawForeground` paints the bar in viewport coordinates so zoom needs no observers.

**Tech Stack:** PySide6 QGraphicsView, Traits/TraitsUI, apptools preferences.

```python
"""Scale-bar maths: the units a calibration can be entered in, the
calibration a drawn line implies, and the round bar a map draws for a
given zoom. Qt-free, so the snapping ladder is testable on its own."""
import math

#: Dropdown order (largest first) and each unit's size in metres.
UNITS = ("m", "cm", "mm", "µm", "nm")
UNIT_METRES = {"m": 1.0, "cm": 1e-2, "mm": 1e-3, "µm": 1e-6, "nm": 1e-9}
DEFAULT_UNIT = "mm"

#: About how wide the drawn bar should be, in screen pixels, before its
#: length is snapped down to a round number.
SCALE_BAR_TARGET_PX = 120.0

#: Shorter calibration drags are a misclick, not a line.
MIN_SCALE_LINE_PX = 5.0

#: A scale bar reads 1, 2 or 5 times a power of ten — never 3.7.
_NICE_STEPS = (1.0, 2.0, 5.0)


def metres_per_pixel(line_px, value, unit):
    """The calibration a line of ``line_px`` called ``value`` ``unit``
    implies. None for a misclick or a non-positive value, so neither
    can poison an existing calibration."""
    if line_px < MIN_SCALE_LINE_PX or value <= 0:
        return None
    return value * UNIT_METRES[unit] / line_px


def format_length(length_m):
    """``length_m`` in the largest unit that still renders it at 1 or
    above: 0.0005 -> '500 µm'. Clamps at nm rather than reading
    '0 nm'."""
    for unit in UNITS:
        value = length_m / UNIT_METRES[unit]
        if value >= 1.0:
            return f"{value:g} {unit}"
    return f"{length_m / UNIT_METRES[UNITS[-1]]:g} {UNITS[-1]}"


def nice_scale(metres_per_screen_px, target_px=SCALE_BAR_TARGET_PX):
    """``(bar_px, label)`` for a bar of about ``target_px``, its length
    snapped DOWN to 1, 2 or 5 times a power of ten so the label always
    reads round. None when there is no usable calibration."""
    if (not math.isfinite(metres_per_screen_px)
            or metres_per_screen_px <= 0):
        return None
    span = metres_per_screen_px * target_px
    decade = 10.0 ** math.floor(math.log10(span))
    length_m = decade
    for step in _NICE_STEPS:
        if step * decade <= span:
            length_m = step * decade
    return length_m / metres_per_screen_px, format_length(length_m)
```

```python
    """Image scale for the on-canvas bar (persisted per experiment).

    metres_per_pixel = Float(0.0)

    value = Float(0.0)
```

```python
"""Canvas layer for the scale calibration: one rubber-banded line whose
length in image pixels is reported on release. Separate from the ROI
layer, which owns creation, editing and selection — a calibration line
is none of those, and is not kept once its length becomes a number."""
import math

from PySide6.QtGui import QColor, QPen
from PySide6.QtWidgets import QGraphicsLineItem

from .scale_bar import MIN_SCALE_LINE_PX

#: Cosmetic (zoom-independent) pen; amber reads on both bright fields
#: and dark raws, and does not clash with the cyan ROI outlines.
SCALE_PEN = QPen(QColor(255, 193, 7), 0)


class ScaleCanvasLayer:
    """Owns the draft calibration line on the image scene."""

    def __init__(self, scene):
        self._scene = scene
        self._draft = None
        self._press_point = None
        self.mode = "pan"
        self.on_line_drawn = lambda length_px: None

    def set_mode(self, mode):
        if mode != "draw_scale":
            self.clear_draft()
        self.mode = mode

    def clear_draft(self):
        if self._draft is not None:
            self._scene.removeItem(self._draft)
        self._draft = None
        self._press_point = None

    def mouse_press(self, scene_point):
        if self.mode != "draw_scale":
            return False
        self._press_point = scene_point
        self._draft = QGraphicsLineItem()
        self._draft.setPen(SCALE_PEN)
        self._scene.addItem(self._draft)
        return True

    def mouse_move(self, scene_point):
        if self._draft is None:
            return False
        self._draft.setLine(self._press_point.x(), self._press_point.y(),
                            scene_point.x(), scene_point.y())
        return True

    def mouse_release(self, scene_point):
        if self._draft is None:
            return False
        length_px = math.hypot(scene_point.x() - self._press_point.x(),
                               scene_point.y() - self._press_point.y())
        self.clear_draft()
        if length_px >= MIN_SCALE_LINE_PX:
            self.on_line_drawn(length_px)
        return True
```

```python
    def set_scale(self, metres_per_pixel, show_bar):
        self._metres_per_pixel = metres_per_pixel
        self._show_scale_bar = show_bar
        self.viewport().update()

    def drawForeground(self, painter, rect):
        """Paint the scale bar as a HUD: reset to viewport pixels, then
        ask nice_scale what a bar of about SCALE_BAR_TARGET_PX should
        read at the current zoom."""
        super().drawForeground(painter, rect)
        if not self._show_scale_bar or self._metres_per_pixel <= 0:
            return
        zoom = self.transform().m11()
        if zoom <= 0:
            return
        scale = nice_scale(self._metres_per_pixel / zoom)
        if scale is None:
            return
        bar_px, label = scale
        painter.save()
        painter.resetTransform()
        painter.setRenderHint(QPainter.RenderHint.Antialiasing, False)
        viewport = self.viewport().rect()
        left = viewport.left() + SCALE_BAR_MARGIN_PX
        bottom = viewport.bottom() - SCALE_BAR_MARGIN_PX
        plate = QRectF(left - 6, bottom - 26, bar_px + 12, 32)
        painter.fillRect(plate, QColor(0, 0, 0, 110))
        painter.setPen(QPen(QColor(255, 255, 255), 2))
        painter.drawLine(left, bottom, left + bar_px, bottom)
        painter.drawLine(left, bottom - 5, left, bottom)
        painter.drawLine(left + bar_px, bottom - 5, left + bar_px, bottom)
        painter.drawText(QRectF(left, bottom - 24, bar_px, 18),
                         Qt.AlignmentFlag.AlignCenter, label)
        painter.restore()
```

```python
"""Offscreen smoke for the scale bar: a calibration reaches the view,
the label re-rounds as the view zooms, and hiding stops the painting."""
import os

os.environ.setdefault("QT_QPA_PLATFORM", "offscreen")

from PySide6.QtCore import QPointF
from PySide6.QtGui import QImage, QPainter
from PySide6.QtWidgets import (
    QApplication, QGraphicsPixmapItem, QGraphicsScene,
)

app = QApplication.instance() or QApplication([])

from fluorescence_controls_ui.image_viewer.scale_bar import nice_scale
from fluorescence_controls_ui.image_viewer.scale_layer import (
    ScaleCanvasLayer,
)
from fluorescence_controls_ui.image_viewer.view import _ImageView
from fluorescence_controls_ui.image_viewer.analysis.roi_canvas_layer import (
    RoiCanvasLayer,
)

scene = QGraphicsScene()
scene.addItem(QGraphicsPixmapItem())
scene.setSceneRect(0, 0, 640, 480)
scale_layer = ScaleCanvasLayer(scene)
view = _ImageView(scene, lambda x, y: None, RoiCanvasLayer(scene),
                  scale_layer)
view.resize(400, 300)
view.show()

# 10 µm per image pixel, drawn at 1:1.
view.resetTransform()
view.set_scale(1e-5, True)


def render(zoom):
    """(painted pixel count, expected label) at this zoom."""
    view.resetTransform()
    view.scale(zoom, zoom)
    image = QImage(400, 300, QImage.Format.Format_ARGB32)
    image.fill(0)
    painter = QPainter(image)
    view.render(painter)
    painter.end()
    painted = sum(1 for y in range(image.height())
                  for x in range(image.width())
                  if image.pixelColor(x, y).alpha())
    scale = nice_scale(1e-5 / view.transform().m11())
    return painted, (scale[1] if scale else None)


for zoom in (1.0, 4.0, 0.25):
    print(f"{zoom}x -> {render(zoom)[1]}")
assert render(1.0)[1] != render(4.0)[1], \
    "zooming must re-round the label"

# Hidden and uncalibrated must both paint strictly less than shown.
shown = render(1.0)[0]
view.set_scale(1e-5, False)
hidden = render(1.0)[0]
view.set_scale(0.0, True)
uncalibrated = render(1.0)[0]
print(f"painted px: shown={shown} hidden={hidden} "
      f"uncalibrated={uncalibrated}")
assert shown > hidden and shown > uncalibrated, \
    "the bar drew nothing, or drew while it should not have"

# The layer still reports a drag through the view's event path.
drawn = []
scale_layer.on_line_drawn = drawn.append
scale_layer.set_mode("draw_scale")
scale_layer.mouse_press(QPointF(0.0, 0.0))
scale_layer.mouse_release(QPointF(50.0, 0.0))
assert drawn == [50.0], drawn
print("smoke passed")
```
